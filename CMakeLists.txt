cmake_minimum_required(VERSION 3.20)
project(SnowWindows VERSION 0.2.19.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(SnowCoreTests tests/core_tests.cpp)
enable_testing()
add_test(NAME Core COMMAND SnowCoreTests)
if(WIN32)
 enable_language(RC)
 add_executable(Snow WIN32 src/main.cpp src/resources.rc)
 target_compile_definitions(Snow PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0a00 WINVER=0x0a00)
 target_include_directories(Snow PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(Snow PRIVATE user32 gdi32 gdiplus comctl32 shell32 ole32 comdlg32 dwmapi wtsapi32 windowscodecs uuid advapi32)
 if(MSVC)
  target_compile_options(Snow PRIVATE /utf-8 /W4)
  set_property(TARGET Snow PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 endif()
 install(TARGETS Snow RUNTIME DESTINATION .)
 install(DIRECTORY assets DESTINATION .)
 install(FILES INSTALL.txt THIRD_PARTY.md DESTINATION .)
endif()
